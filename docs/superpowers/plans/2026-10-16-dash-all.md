# `/dash all` Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `/dash all` to the session-dash mod: a 6-tab pane showing all-time Claude Code usage (models, projects, tools, skills, plugins, MCP, timeline) from local files, with estimated API cost.

**Architecture:** A Python indexer (`bin/history.py`) rolls each transcript file up into `~/.claude/session-dash/history-index.json` and keeps rollups of deleted transcripts. The mod reads that index plus a few small files, and pure TypeScript (`hooks/lib/history*.ts`) aggregates them for the chosen range and project. A view (`hooks/view/history.tsx`) draws the result. The wiring (`hooks/all.tsx`) runs the indexer in the background, loads the files, and holds the pane's state in atoms.

**Tech Stack:** Python 3.12 stdlib (`unittest`), Claude Code mod API (`claude-code`, `claude-code/testing`), TSX against the terminal element table.

**Spec:** `docs/superpowers/specs/2026-10-16-dash-all-design.md`. **Design boards:** https://claude.ai/artifact/KFa58NBVgkH375B1csYjNY

```python
"""List API prices ($/MTok) and token cost, shared by cost.py and history.py.

Cache writes are 1.25x (5m) / 2x (1h) input; reads as listed. On a
subscription this is what the usage would cost at API prices, not a bill.
"""
# $/MTok: input, output, cache read.
PRICE = {
  'opus-5-5': (4, 20, 0.20), 'opus-5': (5, 25, 0.50), 'opus-4-8': (5, 25, 0.50), 'opus-4-7': (5, 25, 0.50),
  'opus-4-6': (5, 25, 0.50), 'opus-4-5': (5, 25, 0.50), 'sonnet-5-5': (2, 10, 0.20), 'sonnet-5': (2, 10, 0.20),
  'sonnet-4-6': (3, 15, 0.30), 'haiku-5-5': (0.10, 0.50, 0.01), 'haiku-4-5': (1, 5, 0.10),
  'fable-5-1': (10, 50, 0.25), 'fable-5': (10, 50, 1.0),
}
def price(model):
    for k in sorted(PRICE, key=len, reverse=True):
        if k in (model or ''): return PRICE[k]
    return None
def cost_parts(u, model):
    """({'in','out','cr','cw'} in $, None), or ({}, model) when the model has no price."""
    p = price(model)
    if not p: return {}, model
    i, o, r = p
    cc = u.get('cache_creation') or {}
    w5 = cc.get('ephemeral_5m_input_tokens'); w1 = cc.get('ephemeral_1h_input_tokens')
    if w5 is None and w1 is None: w5, w1 = u.get('cache_creation_input_tokens', 0), 0
    return {'in': (u.get('input_tokens', 0) or 0) * i / 1e6, 'out': (u.get('output_tokens', 0) or 0) * o / 1e6,
            'cr': (u.get('cache_read_input_tokens', 0) or 0) * r / 1e6,
            'cw': ((w5 or 0) * i * 1.25 + (w1 or 0) * i * 2) / 1e6}, None
def cost(u, model):
    parts, miss = cost_parts(u, model)
    return (sum(parts.values()) if parts else 0.0), miss
```

```python
#!/usr/bin/env python3
"""All-time Claude Code usage, rolled up per transcript file.

  history.py index     update ~/.claude/session-dash/history-index.json, print a one-line JSON report
  history.py rebuild   the same from scratch

Each transcript file becomes one rollup: models with tokens and est. $,
tools, skills, subagents, MCP calls, slash commands, branches, versions and
messages per day. Claude Code deletes old transcripts, so a rollup whose
file is gone is kept, marked pruned: the index doubles as the archive.
Dedup is per file (streamed replies repeat their message id); the data has
no call repeated across files.
"""
import json, os, re, sys, time
from collections import Counter
from datetime import datetime
# Run with python3 -I by the mod, which leaves this folder off sys.path.
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from prices import cost_parts

V = 1
WORKTREE = re.compile(r'^(.*?)/\.?worktrees/([^/]+)')
COMMAND = re.compile(r'<command-name>/?([^<\s]+)</command-name>')


def claude_dir():
    return os.environ.get('SESSION_DASH_CLAUDE_DIR') or os.path.expanduser('~/.claude')


def index_path():
    return os.path.join(claude_dir(), 'session-dash', 'history-index.json')


def split_worktree(cwd):
    m = WORKTREE.match(cwd or '')
    return (m.group(1), m.group(2)) if m else (cwd, None)


def folder_project(path):
    # "-home-me-STS-app" -> "/home/me/STS/app"; lossy for names holding '-', only a fallback
    rel = os.path.relpath(path, os.path.join(claude_dir(), 'projects'))
    return '/' + rel.split(os.sep)[0].lstrip('-').replace('-', '/')


def bump(d, k, n=1):
    d[k] = d.get(k, 0) + n


def text_of(e):
    c = e.get('content')
    if isinstance(c, str): return c
    m = e.get('message')
    c = m.get('content') if isinstance(m, dict) else None
    if isinstance(c, str): return c
    if isinstance(c, list):
        return ' '.join(b.get('text', '') for b in c if isinstance(b, dict) and b.get('type') == 'text')
    return ''


def add_tool(r, b):
    name = b.get('name') or '?'
    i = b.get('input') if isinstance(b.get('input'), dict) else {}
    if name.startswith('mcp__'):
        parts = name.split('__', 2)
        bump(r['mcp'].setdefault(parts[1] if len(parts) > 1 else '?', {}), parts[2] if len(parts) > 2 else '?')
        bump(r['tools'], 'mcp')
        return
    bump(r['tools'], name)
    if name == 'Skill' and i.get('skill'): bump(r['skills'], str(i['skill']))
    if name in ('Agent', 'Task'): bump(r['agents'], str(i.get('subagent_type') or 'general-purpose'))


def parse(path):
    name = os.path.basename(path)
    agent_file = name.startswith('agent-') or 'subagents' in path.split(os.sep)
    r = {'sessionId': None, 'agentFile': agent_file, 'pruned': False, 'cwd': None, 'project': None, 'worktree': None,
         'start': None, 'end': None, 'startDay': None, 'startHour': None, 'versions': {}, 'branches': {},
         'models': {}, 'tools': {}, 'skills': {}, 'agents': {}, 'mcp': {}, 'commands': {}, 'days': {},
         'unpriced': [], 'skipped': 0}
    calls, msgs, tools, cwds, unpriced = set(), set(), set(), Counter(), set()
    with open(path, errors='replace') as fh:
        for line in fh:
            if '"timestamp"' not in line: continue
            try: e = json.loads(line)
            except ValueError: r['skipped'] += 1; continue
            if not isinstance(e, dict) or not isinstance(e.get('timestamp'), str): continue
            try: t = datetime.fromisoformat(e['timestamp'].replace('Z', '+00:00')).astimezone()
            except ValueError: r['skipped'] += 1; continue
            iso = t.isoformat(timespec='seconds')
            if r['start'] is None or iso < r['start']:
                r['start'], r['startDay'], r['startHour'] = iso, t.date().isoformat(), t.hour
            if r['end'] is None or iso > r['end']: r['end'] = iso
            if not r['sessionId'] and isinstance(e.get('sessionId'), str): r['sessionId'] = e['sessionId']
            if isinstance(e.get('cwd'), str) and e['cwd']: cwds[e['cwd']] += 1
            if e.get('version'): bump(r['versions'], str(e['version']))
            if e.get('gitBranch') and e['gitBranch'] != 'HEAD': bump(r['branches'], str(e['gitBranch']))
            typ = e.get('type'); uuid = e.get('uuid')
            if typ in ('user', 'system') and uuid not in msgs:
                for c in COMMAND.findall(text_of(e)): bump(r['commands'], c)
            if typ not in ('user', 'assistant'):
                if uuid: msgs.add(uuid)
                continue
            day = r['days'].setdefault(t.date().isoformat(), {'m': 0, 'c': 0})
            if uuid and uuid not in msgs: msgs.add(uuid); day['m'] += 1
            m = e.get('message')
            if typ != 'assistant' or not isinstance(m, dict): continue
            for b in m.get('content') or []:
                if isinstance(b, dict) and b.get('type') == 'tool_use' and b.get('id') and b['id'] not in tools:
                    tools.add(b['id']); add_tool(r, b)
            u, mid, model = m.get('usage'), m.get('id'), m.get('model')
            if not u or not mid or model in (None, '<synthetic>'): continue
            key = (mid, e.get('requestId'))
            if key in calls: continue
            calls.add(key)
            parts, miss = cost_parts(u, model)
            if miss: unpriced.add(miss)
            usd = sum(parts.values())
            s = r['models'].setdefault(model, {'calls': 0, 'in': 0, 'out': 0, 'cr': 0, 'cw': 0, 'usd': 0.0,
                                               'usdParts': {'in': 0.0, 'out': 0.0, 'cr': 0.0, 'cw': 0.0},
                                               'agentCalls': 0, 'agentUsd': 0.0})
            s['calls'] += 1
            s['in'] += u.get('input_tokens', 0) or 0
            s['out'] += u.get('output_tokens', 0) or 0
            s['cr'] += u.get('cache_read_input_tokens', 0) or 0
            s['cw'] += u.get('cache_creation_input_tokens', 0) or 0
            s['usd'] += usd
            for k, v in parts.items(): s['usdParts'][k] += v
            if agent_file or e.get('isSidechain'): s['agentCalls'] += 1; s['agentUsd'] += usd
            day['c'] += 1
    r['sessionId'] = r['sessionId'] or os.path.splitext(name)[0]
    r['cwd'] = cwds.most_common(1)[0][0] if cwds else None
    r['project'], r['worktree'] = split_worktree(r['cwd']) if r['cwd'] else (folder_project(path), None)
    r['unpriced'] = sorted(unpriced)
    r['days'] = {k: v for k, v in r['days'].items() if v['m'] or v['c']}
    for s in r['models'].values():
        s['usd'] = round(s['usd'], 6); s['agentUsd'] = round(s['agentUsd'], 6)
        s['usdParts'] = {k: round(v, 6) for k, v in s['usdParts'].items()}
    return r
```

```python
def load():
    try:
        with open(index_path()) as fh: j = json.load(fh)

def transcripts():
    out = {}
    for d, _, files in os.walk(os.path.join(claude_dir(), 'projects')):
        for f in files:
            if not f.endswith('.jsonl'): continue
            p = os.path.join(d, f)
            try: st = os.stat(p)
            except OSError: continue
            out[p] = [st.st_size, int(st.st_mtime)]
    return out


def write(j):
    out = index_path()
    os.makedirs(os.path.dirname(out), exist_ok=True)
    tmp = '%s.%d.tmp' % (out, os.getpid())
    with open(tmp, 'w') as fh: json.dump(j, fh, separators=(',', ':'))
    os.replace(tmp, out)


def tz():
    off = datetime.now().astimezone().utcoffset().total_seconds() // 60
    return '%s%02d:%02d' % ('-' if off < 0 else '+', abs(off) // 60, abs(off) % 60)


def run(rebuild=False):
    t0 = time.time()
    old = {} if rebuild else load()
    now = transcripts()
    keep, changed, skipped = {}, 0, 0
    for path, r in old.items():
        if path not in now: r['pruned'] = True; keep[path] = r
        elif [r.get('size'), r.get('mtime')] == now[path]: keep[path] = r
    for path, r in keep.items():
        if path in now: r['pruned'] = False  # a transcript that came back
    for path in sorted(now):
        if path in keep: continue
        # one odd file (a usage that is not a dict, say) must not stop the run
        try: r = parse(path)
        except Exception: skipped += 1; continue
        r['size'], r['mtime'] = now[path]
        skipped += r['skipped']; changed += 1; keep[path] = r
    write({'v': V, 'builtAt': datetime.now().astimezone().isoformat(timespec='seconds'), 'tz': tz(), 'sessions': keep})
    return {'ok': True, 'files': len(now), 'changed': changed, 'skipped': skipped, 'ms': int((time.time() - t0) * 1000)}
```

```ts
// The shapes /dash all reads from disk, and pure parsers of each file's
// text. Every parser takes text and never throws: a file that is missing or
// malformed gives an empty value and the rest of the dashboard still draws.

export type Parts = { in: number; out: number; cr: number; cw: number }

export type ModelRoll = {
  calls: number
  in: number
  out: number
  cr: number
  cw: number
  usd: number
  usdParts: Parts
  agentCalls: number
  agentUsd: number
}

// One transcript file, rolled up by bin/history.py.
export type Rollup = {
  sessionId: string
  agentFile: boolean
  pruned: boolean
  cwd: string | null
  project: string
  worktree: string | null
  start: string | null
  end: string | null
  startDay: string | null
  startHour: number | null
  versions: Record<string, number>
  branches: Record<string, number>
  models: Record<string, ModelRoll>
  tools: Record<string, number>
  skills: Record<string, number>
  agents: Record<string, number>
  mcp: Record<string, Record<string, number>>
  commands: Record<string, number>
  days: Record<string, { m: number; c: number }>
  unpriced: string[]
  skipped: number
}

export type HistoryIndex = { v: 1; builtAt: string; tz: string; sessions: Record<string, Rollup> }

export type Stats = {
  lastDay: string
  firstDay: string | null
  totalSessions: number
  totalMessages: number
  daily: Record<string, { m: number; s: number }>
  modelTokens: Record<string, { total: number; out: number }>
  hours: number[]
  longest: { messages: number; ms: number; at: string } | null
}

export type Prompt = { project: string; at: number }

export type PluginInfo = {
  name: string
  marketplace: string
  skills: string[]
  agents: string[]
  commands: string[]
  mcp: string[]
  background: boolean
}

export type Extras = {
  home: string
  stats: Stats | null
  prompts: Prompt[]
  plugins: PluginInfo[]
  userSkills: string[]
}

const obj = (x: unknown): Record<string, unknown> | null =>
  x && typeof x === 'object' && !Array.isArray(x) ? (x as Record<string, unknown>) : null
const json = (text: string): unknown => {
  try {
    return JSON.parse(text)
  } catch {
    return null
  }
}
const n = (x: unknown) => (typeof x === 'number' && Number.isFinite(x) ? x : 0)

export const parseIndex = (text: string): HistoryIndex | null => {
  const j = obj(json(text))
  return j && j.v === 1 && obj(j.sessions) ? (j as unknown as HistoryIndex) : null
}

export const parseStats = (text: string): Stats | null => {
  const j = obj(json(text))
  if (!j || typeof j.lastComputedDate !== 'string') return null
  const daily: Stats['daily'] = {}
  for (const d of Array.isArray(j.dailyActivity) ? j.dailyActivity : []) {
    const o = obj(d)
    if (o && typeof o.date === 'string') daily[o.date] = { m: n(o.messageCount), s: n(o.sessionCount) }
  }
  const modelTokens: Stats['modelTokens'] = {}
  for (const [k, v] of Object.entries(obj(j.modelUsage) ?? {})) {
    const o = obj(v)
    if (!o) continue
    modelTokens[k] = {
      total: n(o.inputTokens) + n(o.outputTokens) + n(o.cacheReadInputTokens) + n(o.cacheCreationInputTokens),
      out: n(o.outputTokens),
    }
  }
  const hours = Array.from({ length: 24 }, (_, h) => n(obj(j.hourCounts)?.[String(h)]))
  const l = obj(j.longestSession)
  return {
    lastDay: j.lastComputedDate,
    firstDay: typeof j.firstSessionDate === 'string' ? j.firstSessionDate.slice(0, 10) : null,
    totalSessions: n(j.totalSessions),
    totalMessages: n(j.totalMessages),
    daily,
    modelTokens,
    hours,
    longest:
      l && typeof l.timestamp === 'string'
        ? { messages: n(l.messageCount), ms: n(l.duration), at: l.timestamp.slice(0, 10) }
        : null,
  }
}

export const parsePrompts = (text: string): Prompt[] => {
  const out: Prompt[] = []
  for (const line of text.split('\n')) {
    if (!line.trim()) continue
    const o = obj(json(line))
    if (o && typeof o.project === 'string' && typeof o.timestamp === 'number') out.push({ project: o.project, at: o.timestamp })
  }
  return out
}

const WORKTREE = /^(.*?)\/\.?worktrees\/([^/]+)/

export const splitWorktree = (path: string): { project: string; worktree: string | null } => {
  const m = WORKTREE.exec(path)
  return m ? { project: m[1]!, worktree: m[2]! } : { project: path, worktree: null }
}

// installed_plugins.json: {"plugins": {"name@marketplace": [{scope, installPath}]}}
export const pickPlugins = (text: string): { name: string; marketplace: string; path: string }[] => {
  const plugins = obj(obj(json(text))?.plugins)
  if (!plugins) return []
  const out: { name: string; marketplace: string; path: string }[] = []
  for (const [id, entries] of Object.entries(plugins)) {
    const list = (Array.isArray(entries) ? entries : []).map(obj).filter(e => e && typeof e.installPath === 'string')
    const pick = list.find(e => e!.scope === 'user') ?? list[0]
    if (!pick) continue
    const at = id.lastIndexOf('@')
    out.push({ name: at > 0 ? id.slice(0, at) : id, marketplace: at > 0 ? id.slice(at + 1) : '', path: pick.installPath as string })
  }
  return out
}

// What a plugin's folder holds, read by the caller; this decides what it is.
export const pluginInfo = (p: {
  name: string
  marketplace: string
  root: string[]
  skills: string[]
  agents: string[]
  commands: string[]
  mcp: string[]
  pluginJson: string
  mcpJson: string
}): PluginInfo => {
  const mj = obj(json(p.mcpJson))
  const servers = mj ? Object.keys(obj(mj.mcpServers) ?? mj) : []
  const pj = obj(json(p.pluginJson)) ?? {}
  const mcp = [...new Set([...p.mcp, ...servers, ...Object.keys(obj(pj.mcpServers) ?? {})])]
  const countable = p.skills.length + p.agents.length + p.commands.length + mcp.length > 0
  const passive = p.root.includes('hooks') || 'hooks' in pj || 'lspServers' in pj || p.root.includes('bin')
  return {
    name: p.name,
    marketplace: p.marketplace,
    skills: p.skills,
    agents: p.agents,
    commands: p.commands,
    mcp,
    background: !countable && passive,
  }
}

export const reportLine = (stdout: string): string => {
  const j = obj(json(stdout.trim()))
  if (!j || j.ok !== true) return 'indexed'
  return `indexed ${n(j.files)} files · ${(n(j.ms) / 1000).toFixed(1)}s · ${n(j.skipped)} skipped`
}
```

```ts
import type { Extras, HistoryIndex, Parts, Rollup } from './history-data'
import { splitWorktree } from './history-data'

// Everything /dash all draws, for one range and project. Pure: the caller
// reads the files; this only counts. Day series use stats-cache up to its
// last computed day (it survives transcript pruning) and rollups after it;
// with a project filter only rollups can say which project a day belongs to.

export type Range = '7d' | '30d' | 'all'
export type Opts = { range: Range; project: string | null; now: number }
export type TabId = 'overview' | 'models' | 'projects' | 'tools' | 'plugins' | 'timeline'

export type ModelRow = {
  model: string
  calls: number
  in: number
  out: number
  cr: number
  cw: number
  usd: number
  agentCalls: number
  agentUsd: number
  hit: number
  perCall: number
  share: number
}

export type ProjectRow = {
  project: string
  group: string
  prompts: number
  sessions: number
  calls: number
  usd: number
  last: string | null
  worktrees: { name: string; prompts: number; calls: number; usd: number }[]
  branches: { name: string; n: number }[]
  models: { model: string; calls: number }[]
}

export type Counted = { name: string; n: number }

export type HistoryDash = {
  range: Range
  project: string | null
  today: string
  files: number
  pruned: number
  overview: {
    sessions: number
    messages: number
    calls: number
    usd: number
    liveModels: number
    archivedModels: number
    projects: number
    projectsWithTranscripts: number
    days: { day: string; m: number }[]
    retainedOnly: boolean
  }
  models: {
    rows: ModelRow[]
    archived: { model: string; total: number; out: number }[]
    split: Parts | null
    unpriced: string[]
  }
  projects: ProjectRow[]
  tools: {
    tools: Counted[]
    skills: { name: string; source: string; n: number }[]
    unused: string[]
    agents: Counted[]
    commands: Counted[]
  }
  plugins: {
    plugins: { name: string; marketplace: string; status: 'used' | 'background' | 'idle'; uses: number }[]
    mcp: { server: string; kind: 'plugin' | 'claude.ai' | 'local'; calls: number; tools: Counted[] }[]
  }
  timeline: {
    active: number
    streak: number
    longestStreak: number
    avgGap: number | null
    calendar: { day: string; m: number }[]
    hours: number[]
    versions: { v: string; first: string; last: string; current: boolean; stale: boolean }[]
    biggest: { day: string; m: number }[]
    longest: { messages: number; ms: number; at: string } | null
    firstSeen: string | null
  }
}

const DAY = 86_400_000

const tzMinutes = (tz: string) => {
  const m = /^([+-])(\d\d):(\d\d)$/.exec(tz)
  return m ? (m[1] === '-' ? -1 : 1) * (Number(m[2]) * 60 + Number(m[3])) : 0
}
const dayOf = (ms: number, tzMin: number) => new Date(ms + tzMin * 60_000).toISOString().slice(0, 10)
const addDays = (day: string, k: number) => new Date(Date.parse(`${day}T00:00:00Z`) + k * DAY).toISOString().slice(0, 10)
const between = (a: string, b: string) => Math.round((Date.parse(`${b}T00:00:00Z`) - Date.parse(`${a}T00:00:00Z`)) / DAY)

const add = (m: Record<string, number>, k: string, n: number) => {
  m[k] = (m[k] ?? 0) + n
}
const top = (m: Record<string, number>): Counted[] =>
  Object.entries(m)
    .filter(([, n]) => n > 0)
    .map(([name, n]) => ({ name, n }))
    .sort((a, b) => b.n - a.n || a.name.localeCompare(b.name))

const semver = (v: string) => v.split('.').map(x => Number(x) || 0)
const newer = (a: string, b: string) => {
  const [x, y] = [semver(a), semver(b)]
  for (let i = 0; i < Math.max(x.length, y.length); i++) if ((x[i] ?? 0) !== (y[i] ?? 0)) return (x[i] ?? 0) - (y[i] ?? 0)
  return 0
}

const groupOf = (project: string, home: string) => {
  if (!home || !project.startsWith(`${home}/`)) return 'Other'
  const rest = project.slice(home.length + 1).split('/')
  return rest.length >= 2 ? `~/${rest[0]}` : 'Other'
}

const skillSource = (name: string, plugins: Set<string>, user: Set<string>) => {
  const at = name.indexOf(':')
  if (at > 0) {
    const p = name.slice(0, at)
    if (p === 'anthropic-skills') return 'claude.ai'
    return plugins.has(p) ? p : 'built-in'
  }
  return user.has(name) ? 'yours' : 'built-in'
}

export const aggregate = (index: HistoryIndex, x: Extras, o: Opts): HistoryDash => {
  const tzMin = tzMinutes(index.tz)
  const today = dayOf(o.now, tzMin)
  const from = o.range === 'all' ? '' : addDays(today, o.range === '7d' ? -6 : -29)
  const inRange = (d: string) => d >= from && d <= today
  const all = Object.values(index.sessions)
  const ofProject = (r: Rollup) => !o.project || r.project === o.project
  const scope = all.filter(r => ofProject(r) && Object.keys(r.days).some(inRange))
  const prompts = x.prompts
    .map(p => ({ ...p, ...splitWorktree(p.project), day: dayOf(p.at, tzMin) }))
    .filter(p => (!o.project || p.project === o.project) && inRange(p.day))

  // ── day series ────────────────────────────────────────────────────
  const useStats = !o.project && x.stats !== null
  const cut = useStats ? x.stats!.lastDay : ''
  const series: Record<string, { m: number; s: number }> = {}
  if (useStats) for (const [d, v] of Object.entries(x.stats!.daily)) if (inRange(d)) series[d] = { ...v }
  const sessionsByDay: Record<string, Set<string>> = {}
  for (const r of scope) {
    for (const [d, v] of Object.entries(r.days)) {
      if (!inRange(d) || d <= cut) continue
      series[d] = series[d] ?? { m: 0, s: 0 }
      series[d]!.m += v.m
    }
    if (r.startDay && inRange(r.startDay) && r.startDay > cut && !r.agentFile) {
      ;(sessionsByDay[r.startDay] ??= new Set()).add(r.sessionId)
    }
  }
  for (const [d, ids] of Object.entries(sessionsByDay)) (series[d] ??= { m: 0, s: 0 }).s += ids.size
  const days = Object.entries(series)
    .filter(([, v]) => v.m > 0)
    .map(([day, v]) => ({ day, m: v.m }))
    .sort((a, b) => a.day.localeCompare(b.day))

  // ── models ────────────────────────────────────────────────────────
  const mm: Record<string, Omit<ModelRow, 'hit' | 'perCall' | 'share'>> = {}
  const parts: Record<string, Parts> = {}
  const unpriced = new Set<string>()
  for (const r of scope) {
    r.unpriced.forEach(u => unpriced.add(u))
    for (const [id, s] of Object.entries(r.models)) {
      const t = (mm[id] ??= { model: id, calls: 0, in: 0, out: 0, cr: 0, cw: 0, usd: 0, agentCalls: 0, agentUsd: 0 })
      const tp = (parts[id] ??= { in: 0, out: 0, cr: 0, cw: 0 })
      t.calls += s.calls
      t.in += s.in
      t.out += s.out
      t.cr += s.cr
      t.cw += s.cw
      t.usd += s.usd
      t.agentCalls += s.agentCalls
      t.agentUsd += s.agentUsd
      for (const k of ['in', 'out', 'cr', 'cw'] as const) tp[k] += s.usdParts?.[k] ?? 0
    }
  }
  const calls = Object.values(mm).reduce((a, m) => a + m.calls, 0)
  const usd = Object.values(mm).reduce((a, m) => a + m.usd, 0)
  const rows: ModelRow[] = Object.values(mm)
    .map(m => ({
      ...m,
      hit: m.cr + m.cw + m.in > 0 ? m.cr / (m.cr + m.cw + m.in) : 0,
      perCall: m.calls ? m.usd / m.calls : 0,
      share: calls ? m.calls / calls : 0,
    }))
    .sort((a, b) => b.usd - a.usd || b.calls - a.calls)
  const seenModels = new Set(all.flatMap(r => Object.keys(r.models)))
  const archived = useStats
    ? Object.entries(x.stats!.modelTokens)
        .filter(([id]) => !seenModels.has(id))
        .map(([model, t]) => ({ model, total: t.total, out: t.out }))
        .sort((a, b) => b.total - a.total)
    : []
  const split = rows[0] ? { ...parts[rows[0].model]! } : null

  // ── projects ──────────────────────────────────────────────────────
  const pr: Record<string, ProjectRow & { ids: Set<string>; wt: Record<string, { prompts: number; calls: number; usd: number }>; br: Record<string, number>; md: Record<string, number> }> = {}
  const row = (project: string) =>
    (pr[project] ??= { project, group: groupOf(project, x.home), prompts: 0, sessions: 0, calls: 0, usd: 0, last: null, worktrees: [], branches: [], models: [], ids: new Set(), wt: {}, br: {}, md: {} })
  for (const r of scope) {
    const p = row(r.project)
    const c = Object.values(r.models).reduce((a, m) => a + m.calls, 0)
    const u = Object.values(r.models).reduce((a, m) => a + m.usd, 0)
    p.ids.add(r.sessionId)
    p.calls += c
    p.usd += u
    const last = Object.keys(r.days).filter(inRange).sort().at(-1) ?? null
    if (last && (!p.last || last > p.last)) p.last = last
    if (r.worktree) {
      const w = (p.wt[r.worktree] ??= { prompts: 0, calls: 0, usd: 0 })
      w.calls += c
      w.usd += u
    }
    for (const [b, n] of Object.entries(r.branches)) add(p.br, b, n)
    for (const [m, s] of Object.entries(r.models)) add(p.md, m, s.calls)
  }
  for (const q of prompts) {
    const p = row(q.project)
    p.prompts++
    if (q.worktree) (p.wt[q.worktree] ??= { prompts: 0, calls: 0, usd: 0 }).prompts++
    if (!p.last || q.day > p.last) p.last = q.day
  }
  const projects: ProjectRow[] = Object.values(pr)
    .map(({ ids, wt, br, md, ...p }) => ({
      ...p,
      sessions: ids.size,
      worktrees: Object.entries(wt)
        .map(([name, w]) => ({ name, ...w }))
        .sort((a, b) => b.usd - a.usd || b.prompts - a.prompts),
      branches: top(br).slice(0, 3),
      models: top(md).map(c => ({ model: c.name, calls: c.n })),
    }))
    .sort((a, b) => b.usd - a.usd || b.prompts - a.prompts || a.project.localeCompare(b.project))

  // ── tools, skills, agents, commands ───────────────────────────────
  const tools: Record<string, number> = {}
  const skills: Record<string, number> = {}
  const agents: Record<string, number> = {}
  const commands: Record<string, number> = {}
  const mcpCalls: Record<string, Record<string, number>> = {}
  for (const r of scope) {
    for (const [k, n] of Object.entries(r.tools)) add(tools, k, n)
    for (const [k, n] of Object.entries(r.skills)) add(skills, k, n)
    for (const [k, n] of Object.entries(r.agents)) add(agents, k, n)
    for (const [k, n] of Object.entries(r.commands ?? {})) add(commands, k, n)
    for (const [s, t] of Object.entries(r.mcp)) for (const [k, n] of Object.entries(t)) add((mcpCalls[s] ??= {}), k, n)
  }
  const pluginNames = new Set(x.plugins.map(p => p.name))
  const userSkills = new Set(x.userSkills)
  const skillRows = top(skills).map(s => ({ name: s.name, source: skillSource(s.name, pluginNames, userSkills), n: s.n }))
  // a bare name counts for a plugin that ships it; a built-in of the same
  // name would count too, which is rare enough to accept
  const used = (name: string) => (skills[name] ?? 0) + (commands[name] ?? 0)
  const unused = [
    ...x.plugins.flatMap(p => p.skills.map(s => `${p.name}:${s}`)).filter(s => used(s) === 0 && used(s.split(':')[1]!) === 0),
    ...x.userSkills.filter(s => used(s) === 0),
  ].sort()

  // ── plugins and MCP ───────────────────────────────────────────────
  const plugins = x.plugins
    .map(p => {
      let uses = 0
      for (const s of [...p.skills, ...p.commands]) uses += used(`${p.name}:${s}`) + used(s)
      for (const a of p.agents) uses += agents[`${p.name}:${a}`] ?? 0
      for (const s of p.mcp) uses += Object.values(mcpCalls[`plugin_${p.name}_${s}`] ?? {}).reduce((a, n) => a + n, 0)

    })

      return {
        server,
        kind: (server.startsWith('plugin_') ? 'plugin' : server.startsWith('claude_ai_') ? 'claude.ai' : 'local') as 'plugin' | 'claude.ai' | 'local',
        calls: Object.values(t).reduce((a, n) => a + n, 0),
        tools: top(t),
      }
    })
    .sort((a, b) => b.calls - a.calls || a.server.localeCompare(b.server))

  // ── timeline ──────────────────────────────────────────────────────
  const activeDays = days.map(d => d.day)
  const isActive = new Set(activeDays)
  let streak = 0
  for (let d = isActive.has(today) ? today : addDays(today, -1); isActive.has(d); d = addDays(d, -1)) streak++
  let longestStreak = 0
  let run = 0
  let prev = ''
  for (const d of activeDays) {
    run = prev && between(prev, d) === 1 ? run + 1 : 1
    longestStreak = Math.max(longestStreak, run)
    prev = d
  }
  const avgGap = activeDays.length > 1 ? between(activeDays[0]!, activeDays.at(-1)!) / (activeDays.length - 1) : null
  const calStart = addDays(today, -((new Date(`${today}T00:00:00Z`).getUTCDay() + 6) % 7) - 15 * 7) // Monday, 16 weeks back
  const calendar = Array.from({ length: 16 * 7 }, (_, i) => {
    const day = addDays(calStart, i)
    return { day, m: series[day]?.m ?? 0 }
  })
  // stats-cache's hour counts and longest session are all-time totals with
  // no dates, so they only stand in for the 'all' range
  const allTime = useStats && o.range === 'all'
  const hours = allTime ? [...x.stats!.hours] : Array<number>(24).fill(0)
  for (const r of scope) {
    if (r.agentFile || r.startHour === null || !r.startDay || !inRange(r.startDay)) continue
    if (!allTime || r.startDay > cut) hours[r.startHour] = (hours[r.startHour] ?? 0) + 1
  }
  const vs: Record<string, { first: string; last: string }> = {}
  for (const r of scope) {
    for (const v of Object.keys(r.versions)) {
      const s = (vs[v] ??= { first: r.start ?? '', last: r.end ?? '' })
      if (r.start && r.start < s.first) s.first = r.start
      if (r.end && r.end > s.last) s.last = r.end
    }
  }
  const order = Object.keys(vs).sort((a, b) => newer(b, a))
  const current = order[0]
  const versions = order.map(v => ({
    v,
    first: vs[v]!.first.slice(0, 10),
    last: vs[v]!.last.slice(0, 10),
    current: v === current,
    stale: v !== current && current !== undefined && vs[v]!.last > vs[current]!.first,
  }))
  const firstCandidates = [
    useStats ? x.stats!.firstDay : null,
    ...all.map(r => r.startDay),
    ...x.prompts.map(p => dayOf(p.at, tzMin)),
  ].filter((d): d is string => !!d)

  return {
    range: o.range,
    project: o.project,
    today,
    files: all.length,
    pruned: all.filter(r => r.pruned).length,
    overview: {
      sessions: Object.values(series).reduce((a, v) => a + v.s, 0),
      messages: days.reduce((a, d) => a + d.m, 0),
      calls,
      usd,
      liveModels: rows.length,
      archivedModels: archived.length,
      projects: projects.length,
      projectsWithTranscripts: new Set(scope.filter(r => !r.pruned).map(r => r.project)).size,
      days,
      retainedOnly: !useStats,
    },
    models: { rows, archived, split, unpriced: [...unpriced].sort() },
    projects,
    tools: { tools: top(tools), skills: skillRows, unused, agents: top(agents), commands: top(commands) },
    plugins: { plugins, mcp },
    timeline: {
      active: activeDays.length,
      streak,
      longestStreak,
      avgGap,
      calendar,
      hours,
      versions,
      biggest: [...days].sort((a, b) => b.m - a.m).slice(0, 2),
      longest: allTime ? x.stats!.longest : null,
      firstSeen: firstCandidates.sort()[0] ?? null,
    },
  }
}
```

```ts
import type { HistoryDash, TabId } from './history'

// At most one line per tab, and only when it is true by a clear margin:
// a dashboard that always has something to say teaches people to skip it.

const pc = (x: number) => `${Math.round(x * 100)}%`

export const insights = (d: HistoryDash): Partial<Record<TabId, string>> => {
  const out: Partial<Record<TabId, string>> = {}
  const usd = d.overview.usd
  const topModel = d.models.rows[0]
  if (topModel && usd > 0 && topModel.usd / usd >= 0.6) out.overview = `${topModel.model} is ${pc(topModel.usd / usd)} of est. spend`

  const s = d.models.split
  const sTotal = s ? s.in + s.out + s.cr + s.cw : 0
  if (topModel && s && sTotal > 0 && s.cr / sTotal >= 0.5) {
    out.models = `Re-reading cached context is ${pc(s.cr / sTotal)} of ${topModel.model}’s cost. Shorter sessions or /compact cut it most.`
  }

  const n = (name: string) => d.tools.tools.find(t => t.name === name)?.n ?? 0
  const toolTotal = d.tools.tools.reduce((a, t) => a + t.n, 0)
  if (n('Edit') === 0 && n('Bash') > 100) {
    out.tools = `0 Edit calls and ${n('Bash')} Bash calls: file changes are going through Bash.`
  } else if (toolTotal > 0 && d.tools.tools[0]!.n / toolTotal >= 0.6) {
    out.tools = `${d.tools.tools[0]!.name} is ${pc(d.tools.tools[0]!.n / toolTotal)} of tool calls`
  }

  const idle = d.plugins.plugins.filter(p => p.status === 'idle').length
  if (idle >= 5) out.plugins = `${idle} of ${d.plugins.plugins.length} plugins have no recorded use.`

  const topProject = d.projects[0]
  if (topProject && usd > 0 && topProject.usd / usd >= 0.6) out.projects = `${topProject.project} is ${pc(topProject.usd / usd)} of est. spend`

  const hours = d.timeline.hours
  const hTotal = hours.reduce((a, h) => a + h, 0)
  const peak = hours.indexOf(Math.max(...hours))
  if (hTotal > 0 && hours[peak]! / hTotal >= 0.5) {
    const hh = (h: number) => `${String(h % 24).padStart(2, '0')}:00`
    out.timeline = `${pc(hours[peak]! / hTotal)} of sessions start between ${hh(peak)} and ${hh(peak + 1)}`
  }
  return out
}
```

```ts
const EIGHTHS = ['', '▏', '▎', '▍', '▌', '▋', '▊', '▉']

// A bar `w` cells wide filled to `frac`, in eighth-cell steps, padded so
// columns after it line up.
export const hbar = (frac: number, w: number): string => {
  const f = Math.min(1, Math.max(0, Number.isFinite(frac) ? frac : 0))
  const eighths = Math.round(f * w * 8)
  const full = Math.floor(eighths / 8)
  const s = '█'.repeat(full) + (full < w ? EIGHTHS[eighths % 8]! : '')
  return s.padEnd(w)
}

export const num = (n: number): string => String(Math.round(n)).replace(/\B(?=(\d{3})+(?!\d))/g, ',')
```

```tsx
// the last line: a Python traceback ends with the error itself
const lastLine = (s: string) => s.trim().split('\n').at(-1)?.slice(0, 120) ?? ''

const reindex = async ($: EngineInterface) => {
  if (hist.isRunning || !live.home) return
  hist.isRunning = true
  try {
    const r = await $.process.run(['/usr/bin/env', 'python3', '-I', `${binDir(live.home)}/history.py`, 'index'], { timeoutMs: 120_000 })
    const line = r.exitCode === 0 ? reportLine(r.stdout) : `index failed: ${lastLine(r.stderr || r.stdout) || `exit ${r.exitCode}`}`
    await update($, statusAtom, () => line)
  } catch (err) {
    await update($, statusAtom, () => `index failed: ${lastLine(String(err))}`)
  } finally {
    hist.isRunning = false
  }
  await load($)
  await update($, revAtom, n => n + 1)
}

export const toggleAll = async ($: EngineInterface): Promise<{ text: string }> => {
  const mine = (await $.ui.panes()).find(p => p.id === ALL_PANE)
  if (mine?.isShown) {
    await $.ui.close({ id: ALL_PANE })
    return { text: 'Usage closed.' }
  }
  // a reload or another home (tests) starts from disk again
  if (hist.home !== live.home) Object.assign(hist, { home: live.home, index: null, extras: null, isRunning: false })
  if (!hist.index && live.home) await load($)
  swallow(reindex($))
  const opened = await $.ui.open({ id: ALL_PANE, title: 'Usage', columns: 110, focus: true, closeOnEscape: true })
  return { text: opened.isPlaced ? 'Usage open.' : 'Usage waits for a wider terminal.' }
}

const RANGES: Range[] = ['all', '7d', '30d']

export const registerAll = (on: On) => {
  on('ui.render', { component: 'Pane', requestId: ALL_PANE }, async ($, e, next) => {
    if (e.surface !== 'terminal') return next(e)
    const El = $.ui.resolve(e)
    await read($, revAtom)
    const tab = (await read($, tabAtom)) as TabId
    const range = (await read($, rangeAtom)) as Range
    const project = await read($, projectAtom)
    const status = await read($, statusAtom)
    const d = hist.index && hist.extras ? aggregate(hist.index, hist.extras, { range, project, now: await $.clock.now() }) : null
    return (
      <History
        E={El}
        w={e.props.bodyColumns}
        d={d}
        tab={tab}
        status={status}
        insight={d ? insights(d)[tab] : undefined}
        home={live.home}
        act={{
          tab: id => swallow(update($, tabAtom, () => id)),
          range: () => swallow(update($, rangeAtom, r => RANGES[(RANGES.indexOf(r as Range) + 1) % RANGES.length]!)),
          project: p => swallow(update($, projectAtom, () => p)),
        }}
      />
    )
  }).catch(($, e, next) => next(e))
}
```
